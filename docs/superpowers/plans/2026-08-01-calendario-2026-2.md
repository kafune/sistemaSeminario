- [ ] **Step 2: Executar a reconciliação idempotente em uma transação**

Run:

```bash
docker compose exec -T backend python - <<'PY'
from datetime import date, time

from sqlalchemy import func, select

from app.database import SessionLocal
from app.models import Aula, DocTurma, Materia, MatProf, Professor, Turma


def unico(db, modelo, criterio, rotulo):
    itens = list(db.scalars(select(modelo).where(criterio)))
    if len(itens) != 1:
        raise RuntimeError(f"{rotulo}: esperado 1 registro, encontrados {len(itens)}")
    return itens[0]


def materia(db, nome, area=None, apelido=None, observa=None):
    itens = list(db.scalars(select(Materia).where(func.trim(Materia.NOME) == nome)))
    if len(itens) > 1:
        raise RuntimeError(f"Matéria duplicada: {nome}")
    item = itens[0] if itens else Materia(NOME=nome)
    if not itens:
        db.add(item)
    item.NOME = nome
    if area is not None:
        item.area = area
    if apelido is not None:
        item.APELIDO = apelido
    if observa is not None:
        item.observa = observa
    db.flush()
    return item


def garantir_atuacao(db, professor, disciplina):
    existente = db.scalar(
        select(MatProf).where(
            MatProf.cod_pro == professor.cod_pro,
            MatProf.cod_mat == disciplina.cod_mat,
        )
    )
    if existente:
        return
    maior = db.scalar(
        select(func.coalesce(func.max(MatProf.seq_mp), 0)).where(
            MatProf.cod_pro == professor.cod_pro
        )
    )
    db.add(MatProf(cod_pro=professor.cod_pro, cod_mat=disciplina.cod_mat, seq_mp=int(maior) + 1))


def garantir_vinculo(db, turma, disciplina, professor, inicio, fim):
    criterio = (
        (DocTurma.cod_tur == turma.cod_tur)
        & (DocTurma.cod_mat == disciplina.cod_mat)
        & (DocTurma.cod_pro == professor.cod_pro)
        & (DocTurma.Ano == "2026")
        & (DocTurma.semestre == "2")
    )
    itens = list(db.scalars(select(DocTurma).where(criterio)))
    if len(itens) > 1:
        raise RuntimeError(
            f"Vínculo duplicado: {turma.nome} / {disciplina.NOME} / {professor.nome}"
        )
    vinculo = itens[0] if itens else DocTurma(
        cod_tur=turma.cod_tur,
        cod_mat=disciplina.cod_mat,
        cod_pro=professor.cod_pro,
        Ano="2026",
        semestre="2",
    )
    if not itens:
        db.add(vinculo)
    vinculo.dat_ini = inicio
    vinculo.dat_fim = fim
    db.flush()
    return vinculo


def garantir_aula(db, turma, vinculo, dia):
    itens = list(
        db.scalars(
            select(Aula)
            .join(DocTurma, DocTurma.id == Aula.docturma_id)
            .where(
                DocTurma.cod_tur == turma.cod_tur,
                Aula.data == dia,
                Aula.hora_inicio == time(9, 0),
            )
        )
    )
    if len(itens) > 1:
        raise RuntimeError(f"Mais de uma aula encontrada em {dia:%d/%m/%Y}")
    if itens and itens[0].docturma_id != vinculo.id:
        raise RuntimeError(f"Aula conflitante encontrada em {dia:%d/%m/%Y}")
    aula = itens[0] if itens else Aula(
        docturma_id=vinculo.id,
        data=dia,
        hora_inicio=time(9, 0),
    )
    if not itens:
        db.add(aula)
    aula.hora_fim = time(17, 0)
    aula.status = "AGENDADA"


db = SessionLocal()
try:
    with db.begin():
        sabado = unico(db, Turma, Turma.nome == "Sabado", "Turma Sabado")
        noturno = unico(db, Turma, Turma.nome == "Noturno", "Turma Noturno")

        professores = {
            nome: unico(db, Professor, Professor.nome == nome, f"Professor {nome}")
            for nome in (
                "Adarlei Martins",
                "André Oliveira",
                "Eduardo Franco Bernardes",
                "Erisvaldo Verissimo da Silva",
                "Jeferson Alcântara",
                "Vitor Gadelha Gomes de Sá",
            )
        }

        ceb = unico(
            db,
            Materia,
            func.trim(Materia.NOME) == "Comunicação e Ensino Bíblico",
            "Matéria Comunicação e Ensino Bíblico",
        )
        abvf = unico(
            db,
            Materia,
            func.trim(Materia.NOME) == "Aconselhamento Bíblico e Vida Familiar",
            "Matéria Aconselhamento Bíblico e Vida Familiar",
        )
        unico(
            db,
            Materia,
            func.trim(Materia.NOME) == "Hermenêutica e Métodos de Estudo Bíblico",
            "Matéria Hermenêutica e Métodos de Estudo Bíblico",
        )
        ts2 = materia(
            db,
            "Teologia Sistemática II",
            "Teologia Sistemática",
            "TS II",
            "Cristologia, Antropologia e Hamartiologia",
        )
        ts3 = materia(
            db,
            "Teologia Sistemática III",
            "Teologia Sistemática",
            "TS III",
            "Angelologia, Eclesiologia e Escatologia",
        )

        ceb_noturno = unico(
            db,
            DocTurma,
            (DocTurma.cod_tur == noturno.cod_tur)
            & (DocTurma.cod_mat == ceb.cod_mat)
            & (DocTurma.cod_pro == professores["Erisvaldo Verissimo da Silva"].cod_pro),
            "Vínculo noturno de Comunicação e Ensino Bíblico",
        )
        abvf_noturno = unico(
            db,
            DocTurma,
            (DocTurma.cod_tur == noturno.cod_tur)
            & (DocTurma.cod_mat == abvf.cod_mat)
            & (DocTurma.cod_pro == professores["Erisvaldo Verissimo da Silva"].cod_pro),
            "Vínculo noturno de Aconselhamento Bíblico e Vida Familiar",
        )
        ceb_noturno.dat_fim = date(2026, 10, 21)
        abvf_noturno.dat_ini = date(2026, 10, 28)
        abvf_noturno.dat_fim = date(2026, 12, 9)

        aula_2810 = list(
            db.scalars(
                select(Aula)
                .join(DocTurma, DocTurma.id == Aula.docturma_id)
                .where(
                    DocTurma.cod_tur == noturno.cod_tur,
                    Aula.data == date(2026, 10, 28),
                    Aula.hora_inicio == time(19, 15),
                )
            )
        )
        if len(aula_2810) != 1:
            raise RuntimeError(f"Aula noturna de 28/10: esperado 1 registro, encontrados {len(aula_2810)}")
        aula_2810[0].docturma_id = abvf_noturno.id

        agenda = [
            (ts2, "Adarlei Martins", date(2026, 8, 8)),
            (ts2, "Eduardo Franco Bernardes", date(2026, 8, 15)),
            (ts2, "Erisvaldo Verissimo da Silva", date(2026, 8, 29)),
            (ceb, "André Oliveira", date(2026, 9, 19)),
            (ceb, "Erisvaldo Verissimo da Silva", date(2026, 9, 26)),
            (ceb, "André Oliveira", date(2026, 10, 24)),
            (ts3, "Jeferson Alcântara", date(2026, 11, 14)),
            (ts3, "Erisvaldo Verissimo da Silva", date(2026, 11, 28)),
            (ts3, "Jeferson Alcântara", date(2026, 12, 12)),
        ]
        faixas = {}
        for disciplina, nome_professor, dia in agenda:
            chave = (disciplina.cod_mat, nome_professor)
            faixas.setdefault(chave, []).append(dia)

        vinculos = {}
        for (cod_mat, nome_professor), dias in faixas.items():
            disciplina = ts2 if cod_mat == ts2.cod_mat else ts3 if cod_mat == ts3.cod_mat else ceb
            professor = professores[nome_professor]
            garantir_atuacao(db, professor, disciplina)
            vinculos[(cod_mat, nome_professor)] = garantir_vinculo(
                db, sabado, disciplina, professor, min(dias), max(dias)
            )

        for disciplina, nome_professor, dia in agenda:
            garantir_aula(db, sabado, vinculos[(disciplina.cod_mat, nome_professor)], dia)

        db.flush()
        inicio = date(2026, 8, 1)
        fim = date(2026, 12, 31)
        for turma, esperado in ((noturno, 37), (sabado, 9)):
            quantidade = db.scalar(
                select(func.count())
                .select_from(Aula)
                .join(DocTurma, DocTurma.id == Aula.docturma_id)
                .where(DocTurma.cod_tur == turma.cod_tur, Aula.data.between(inicio, fim))
            )
            if quantidade != esperado:
                raise RuntimeError(f"{turma.nome}: esperadas {esperado} aulas, encontradas {quantidade}")
finally:
    db.close()

print("Calendário 2026.2 reconciliado com sucesso.")
PY
```

Expected: `Calendário 2026.2 reconciliado com sucesso.` e exit code 0. Qualquer referência ausente, duplicidade ou conflito reverte toda a transação.
